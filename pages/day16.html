The 'Why' — What This Capstone Builds

A Real WLAN Scanner from Scratch

Tools like Kismet, inSSIDer, and Acrylic Wi-Fi are black boxes. You click scan, wait, and get a list of APs. Today you build the same thing from first principles — and because you own every line, you can customise it for any test scenario: add rogue AP detection, RSSI trending, security auditing, or channel utilisation analysis.

Architecture: three concurrent components.
Hopper thread — cycles channels every 150ms using iw.
Sniffer thread — calls sniff() to capture beacons on the current channel.
AP Registry — a thread-safe shared dict accumulating all discovered APs with RSSI history.

# =============================================================
# Day 16 Capstone: Multi-Channel AP Scanner
# Combines: channel hopper + scapy sniffer + AP registry + CSV
# Requires: root/sudo + monitor-mode capable adapter
# =============================================================

import threading, time, csv, sys, subprocess
from pathlib import Path
from collections import defaultdict
from scapy.all import sniff, Dot11, Dot11Beacon, Dot11Elt, RadioTap
import datetime

CHANNELS = [1, 6, 11, 36, 40, 44, 48, 149, 153, 157, 161]
IFACE    = "wlan0mon"

# Thread-safe AP registry using a lock
# Structure: {bssid: {ssid, rssi_samples, channel, first_seen, last_seen}}
ap_registry = {}
registry_lock = threading.Lock()   # Prevents concurrent write corruption
Block 2 — Beacon Handler with RSSI History

Instead of storing just the latest RSSI, the registry keeps a list of all RSSI samples. This enables min/max/average RSSI reporting — much more useful for coverage analysis than a single snapshot value.

Python 
def parse_beacon_ssid(pkt):
    """Extract SSID string from a beacon frame's Information Elements."""
    ie = pkt.getlayer(Dot11Elt)
    while ie:
        if ie.ID == 0:
            return ie.info.decode("utf-8", errors="replace").strip()
        ie = ie.payload.getlayer(Dot11Elt)
    return ""

def beacon_handler(pkt):
    """
    Process a beacon frame and update the AP registry.
    Uses a lock to safely update the shared registry from the sniffer thread.
    """
    if not pkt.haslayer(Dot11Beacon):
        return

    bssid = pkt[Dot11].addr3
    ssid  = parse_beacon_ssid(pkt)
    rssi  = getattr(pkt.getlayer(RadioTap), "dBm_AntSignal", None)
    now   = datetime.datetime.now().isoformat(timespec="seconds")

    with registry_lock:   # Acquire lock before modifying shared dict
        if bssid not in ap_registry:
            ap_registry[bssid] = {
                "ssid": ssid, "bssid": bssid,
                "rssi_samples": [], "channel": None,
                "first_seen": now, "last_seen": now
            }
        entry = ap_registry[bssid]
        if rssi is not None:
            entry["rssi_samples"].append(rssi)
        entry["last_seen"] = now
Block 3 — Hopper Thread and Sniffer
Python 
def channel_hopper(iface, channels, dwell, stop_event):
    """Hop channels in a loop until stop_event is set."""
    idx = 0
    while not stop_event.is_set():
        ch = channels[idx % len(channels)]
        subprocess.run(["iw", "dev", iface, "set", "channel", str(ch)],
                       capture_output=True)
        time.sleep(dwell)
        idx += 1

def run_scanner(iface, duration=30):
    """
    Run the multi-channel scanner for `duration` seconds.
    Starts hopper thread, then sniffs in the main thread.
    """
    stop_event = threading.Event()
    hopper     = threading.Thread(
        target=channel_hopper,
        args=(iface, CHANNELS, 0.15, stop_event),
        daemon=True
    )
    hopper.start()
    print(f"[+] Scanning {len(CHANNELS)} channels for {duration}s on {iface}")

    try:
        sniff(iface=iface, prn=beacon_handler,
              lfilter=lambda p: p.haslayer(Dot11Beacon),
              timeout=duration, store=False)
    finally:
        stop_event.set()
        print(f"[+] Scan complete. {len(ap_registry)} APs discovered.")

def write_report(registry, out_path):
    """Write AP registry to a CSV report with RSSI statistics."""
    rows = []
    for bssid, info in registry.items():
        samples = info["rssi_samples"]
        avg_rssi = sum(samples) / len(samples) if samples else None
        rows.append({
            "ssid":       info["ssid"],
            "bssid":      bssid,
            "rssi_avg":   f"{avg_rssi:.1f}" if avg_rssi else "N/A",
            "rssi_min":   min(samples) if samples else "N/A",
            "rssi_max":   max(samples) if samples else "N/A",
            "beacons":    len(samples),
            "first_seen": info["first_seen"],
            "last_seen":  info["last_seen"],
        })

    # Sort by average RSSI descending (strongest first)
    rows.sort(key=lambda r: float(r["rssi_avg"]) if r["rssi_avg"] != "N/A" else -999,
              reverse=True)

    with open(out_path, "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=rows[0].keys())
        writer.writeheader()
        writer.writerows(rows)
    print(f"[CSV] Report: {out_path}  ({len(rows)} APs)")

def main():
    out = Path("reports")
    out.mkdir(exist_ok=True)
    run_scanner(IFACE, duration=30)
    if ap_registry:
        ts   = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        path = out / f"ap_scan_{ts}.csv"
        write_report(ap_registry, path)
        # Print top 5 APs
        print("\n[Top APs by RSSI]")
        for bssid, info in list(ap_registry.items())[:5]:
            s = info["rssi_samples"]
            avg = sum(s)/len(s) if s else 0
            print(f"  {info['ssid']:25s} {bssid}  avg {avg:.1f} dBm")
    sys.exit(0)

if __name__ == "__main__":
    main()
