DHCP Verification —
Confirming Your IP Is Correct

A successful association doesn't mean a correct IP. DHCP misconfigurations, exhausted scopes, and APIPA fallbacks are common post-connection failures. The 'Why' — Conceptual Breakdown

Why DHCP Verification Matters

"Connected to Wi-Fi" means the association succeeded. It does not mean you can actually use the network. A client might associate successfully but receive: an APIPA address (169.254.x.x) because the DHCP scope is exhausted; an address from the wrong subnet due to a misconfigured VLAN; a valid address but no default gateway due to a DHCP server bug; a valid address with a DNS server that doesn't respond. Each of these is a distinct failure that requires a different fix.

DHCP validation is a 5-layer check, not a binary test.
Check 1: Is an IP assigned (and not APIPA)? Check 2: Is the subnet mask correct? Check 3: Is a default gateway assigned? Check 4: Is the gateway reachable (ping)? Check 5: Is DNS resolving correctly? All 5 must pass for the connection to be considered fully operational.

Block 1 — get_ip_config() with netifaces

netifaces gives us structured access to interface addresses without parsing command output. It handles all address families (IPv4, IPv6, MAC) and works consistently across Linux, macOS, and Windows.

Python Day22_dhcp_verify.py 
# =============================================================
# Day 22: DHCP Verification Post-Connection
# =============================================================

import netifaces
import subprocess
import re
import ipaddress
from pathlib import Path

def get_ip_config(interface="wlan0"):
    """
    Read full IP configuration for an interface using netifaces.
    Returns a dict with ip, netmask, gateway, mac.
    """
    config = {"interface": interface, "ip": None, "netmask": None,
              "gateway": None, "mac": None}

    try:
        addrs = netifaces.ifaddresses(interface)

        # IPv4 address and netmask
        ipv4_list = addrs.get(netifaces.AF_INET, [])
        if ipv4_list:
            config["ip"]      = ipv4_list[0].get("addr")
            config["netmask"] = ipv4_list[0].get("netmask")

        # MAC address
        mac_list = addrs.get(netifaces.AF_LINK, [])
        if mac_list:
            config["mac"] = mac_list[0].get("addr")

    except ValueError:
        pass   # Interface doesn't exist

    # Default gateway (from routing table)
    gateways = netifaces.gateways()
    default_gw = gateways.get("default", {}).get(netifaces.AF_INET)
    if default_gw:
        gw_ip, gw_iface = default_gw[0], default_gw[1]
        if gw_iface == interface:
            config["gateway"] = gw_ip

    return config

cfg = get_ip_config("wlan0")
print(f"[IP Config] {cfg}")
Block 2 — validate_dhcp() 5-Check Chain

Each check builds on the previous. If a check fails, subsequent checks that depend on it are skipped rather than failing with a misleading error. This produces a clear diagnostic chain.

Python 
def ping_once(host, timeout=2):
    """Return True if host responds to a single ICMP ping."""
    result = subprocess.run(
        ["ping", "-c", "1", "-W", str(timeout), host],
        capture_output=True
    )
    return result.returncode == 0

def validate_dhcp(interface="wlan0", expected_subnet=None):
    """
    Run 5-check DHCP validation chain on an interface.
    Returns a list of (check_name, passed, reason) tuples.
    """
    checks = []
    cfg = get_ip_config(interface)

    # Check 1: IP assigned (and not APIPA 169.254.x.x)
    ip = cfg.get("ip")
    if ip and not ip.startswith("169.254."):
        checks.append(("IP assigned", True, ip))
    else:
        checks.append(("IP assigned", False,
                        f"No valid IP — got: {ip or 'None'} (APIPA/missing)"))
        return checks   # Cannot continue without an IP

    # Check 2: Subnet mask is valid (not /0 or /32)
    netmask = cfg.get("netmask", "")
    try:
        network = ipaddress.IPv4Network(f"{ip}/{netmask}", strict=False)
        prefix  = network.prefixlen
        ok = 8 <= prefix <= 30   # Reasonable range for WLAN clients
        checks.append(("Subnet valid", ok,
                        f"/{prefix} ({netmask})"))
    except ValueError as e:
        checks.append(("Subnet valid", False, str(e)))

    # Check 3: Expected subnet match (optional)
    if expected_subnet:
        try:
            expected = ipaddress.IPv4Network(expected_subnet, strict=False)
            in_subnet = ipaddress.IPv4Address(ip) in expected
            checks.append(("Correct subnet", in_subnet,
                            f"{ip} in {expected_subnet}: {in_subnet}"))
        except ValueError:
            checks.append(("Correct subnet", False, "Invalid expected_subnet"))

    # Check 4: Gateway assigned and reachable
    gw = cfg.get("gateway")
    if gw:
        reachable = ping_once(gw)
        checks.append(("Gateway reachable", reachable,
                        f"{gw} ping: {'OK' if reachable else 'FAIL'}"))
    else:
        checks.append(("Gateway reachable", False, "No gateway assigned"))

    # Check 5: Basic connectivity (ping well-known host)
    net_ok = ping_once("8.8.8.8", timeout=3)
    checks.append(("Internet reachable", net_ok,
                    "8.8.8.8 ping: " + ("OK" if net_ok else "FAIL")))

    return checks

print("\n[DHCP Validation]")
results = validate_dhcp("wlan0", expected_subnet="192.168.1.0/24")
for name, ok, reason in results:
    icon = "✓" if ok else "✗"
    print(f"  [{icon}] {name:25s} {reason}")
Block 3 — read_dhcp_lease() from NetworkManager

NetworkManager stores DHCP lease information in /var/lib/NetworkManager/. Reading these files gives you the full lease details: server IP, lease time, DNS servers, and domain name — all without running any additional commands.

Python 
def read_dhcp_lease(interface="wlan0"):
    """
    Read DHCP lease details from NetworkManager's lease file.
    NM stores leases in /var/lib/NetworkManager/dhclient-<iface>.conf
    or in newer versions as internal-dhcp-<uuid>.lease
    """
    nm_dir = Path("/var/lib/NetworkManager")
    if not nm_dir.exists():
        return None

    # Find lease files for this interface
    for lease_file in nm_dir.glob(f"*{interface}*.lease"):
        lease = {}
        try:
            with open(lease_file) as f:
                for line in f:
                    line = line.strip()
                    # Lease files have: option name value;
                    m = re.match(r"option\s+(\S+)\s+(.+);", line)
                    if m:
                        lease[m.group(1)] = m.group(2).strip('"')
                    # Also: fixed-address IP;
                    m2 = re.match(r"fixed-address\s+([\d.]+);", line)
                    if m2:
                        lease["fixed-address"] = m2.group(1)
        except (IOError, PermissionError):
            continue
        if lease:
            return lease
    return None

lease = read_dhcp_lease("wlan0")
if lease:
    print(f"\n[DHCP Lease]")
    for key, val in lease.items():
        print(f"  {key:30s}: {val}")
