The 'Why' — Conceptual Breakdown

Why Timing Data Matters

Enterprise SLA requirements often specify maximum connection times. A deployment acceptance test might require: total connection time < 5 seconds, DHCP assignment < 3 seconds, and roaming time < 150ms. These requirements can only be verified with precise timing instrumentation.

time.perf_counter() is the right tool for phase timing.
time.time() can jump forward or backward when the system clock is adjusted (NTP sync). time.perf_counter() is a monotonic timer that never goes backward and has sub-microsecond resolution — the right choice for measuring connection phases in milliseconds.

Connection Phases to Measure

The measurable phases in a WPA2 connection are: (1) Scan latency — time to find the target SSID; (2) Authentication — Open System authentication exchange; (3) Association — 802.11 association request/response; (4) 4-Way Handshake — WPA key exchange (usually the longest crypto phase); (5) DHCP — time from association complete to valid IP assigned. Total = sum of all phases.

Block 1 — AssociationTimer Class

The timer uses a dict of named marks and computes elapsed time between consecutive marks. This generalises to any multi-phase timing scenario.

Python Day25_assoc_timing.py 
# =============================================================
# Day 25: Association Timing Measurement
# =============================================================

import time
import subprocess
import re
import netifaces
from pathlib import Path

class AssociationTimer:
    """
    Instruments a connection sequence by recording named time marks.
    Computes elapsed time between consecutive marks.
    """

    def __init__(self):
        self._marks = {}   # {phase_name: perf_counter timestamp}
        self._order = []   # Preserve insertion order (Python 3.7+)

    def mark(self, phase_name):
        """Record the current time for a named phase."""
        t = time.perf_counter()
        self._marks[phase_name] = t
        self._order.append(phase_name)
        return t

    def elapsed_ms(self, start_phase, end_phase):
        """Return elapsed time in ms between two named marks."""
        if start_phase not in self._marks or end_phase not in self._marks:
            return None
        return round(
            (self._marks[end_phase] - self._marks[start_phase]) * 1000, 1
        )

    def total_ms(self):
        """Return total elapsed time from first to last mark in ms."""
        if len(self._order) < 2:
            return None
        return self.elapsed_ms(self._order[0], self._order[-1])

    def report(self):
        """Print a formatted timing report."""
        print("\n[Association Timing Report]")
        print(f"  {'Phase':25s}  {'Duration':>10s}")
        print("  " + "-" * 38)
        for i in range(1, len(self._order)):
            phase = self._order[i]
            prev  = self._order[i - 1]
            ms    = self.elapsed_ms(prev, phase)
            bar   = "█" * min(30, int((ms or 0) / 50))
            print(f"  {phase:25s}  {ms:9.1f}ms  {bar}")
        print("  " + "-" * 38)
        print(f"  {'TOTAL':25s}  {self.total_ms():9.1f}ms")
Block 2 — timed_connect() Measuring Each Phase

This function wraps the connection process from Day 20, inserting timer marks at each observable phase transition. We can't mark the 4-way handshake directly (it happens inside wpa_supplicant), so we mark "association requested" and "IP assigned" — and infer the handshake time from the difference.

Python 
def get_ip(interface):
    """Return current IPv4 address or None."""
    try:
        addrs = netifaces.ifaddresses(interface)
        ipv4  = addrs.get(netifaces.AF_INET, [{}])
        ip = ipv4[0].get("addr", "")
        return ip if ip and not ip.startswith("169.254.") else None
    except:
        return None

def timed_connect(ssid, password, interface="wlan0", timeout=30):
    """
    Connect to WPA2 AP with timing instrumentation on each phase.
    Returns: (timer: AssociationTimer, result: dict)
    """
    timer = AssociationTimer()

    # Phase 0: Start
    timer.mark("start")

    # Phase 1: Scan
    scan = subprocess.run(
        ["nmcli", "dev", "wifi", "list", "--rescan", "yes"],
        capture_output=True, text=True, timeout=20
    )
    timer.mark("scan_complete")

    if ssid not in scan.stdout:
        return timer, {"success": False, "error": f"'{ssid}' not found in scan"}

    # Phase 2: Disconnect (ensure clean state)
    subprocess.run(["nmcli", "dev", "disconnect", interface], capture_output=True)
    timer.mark("disconnect_complete")

    # Phase 3: Trigger connection (nmcli handles auth + assoc + handshake)
    subprocess.run(
        ["nmcli", "--wait", "0", "dev", "wifi", "connect",
         ssid, "password", password, "ifname", interface],
        capture_output=True, text=True
    )
    timer.mark("connect_triggered")

    # Phase 4: Poll until association (check for active SSID)
    deadline = time.perf_counter() + timeout
    while time.perf_counter() < deadline:
        result = subprocess.run(
            ["nmcli", "-t", "-f", "ACTIVE,SSID", "dev", "wifi"],
            capture_output=True, text=True
        )
        if f"yes:{ssid}" in result.stdout:
            timer.mark("associated")
            break
        time.sleep(0.1)
    else:
        return timer, {"success": False, "error": "Association timeout"}

    # Phase 5: Poll until IP assigned
    while time.perf_counter() < deadline:
        ip = get_ip(interface)
        if ip:
            timer.mark("ip_assigned")
            return timer, {"success": True, "ip": ip}
        time.sleep(0.2)

    return timer, {"success": False, "error": "DHCP timeout"}
Block 3 — check_against_baseline()

A baseline is a set of threshold values for each phase, typically derived from the AP vendor's specification or from a known-good reference measurement.

Python 
DEFAULT_BASELINE = {
    "scan_complete":       3000,   # Max 3s for scan
    "associated":          5000,   # Max 5s for full association
    "ip_assigned":        10000,   # Max 10s total for DHCP
}

def check_against_baseline(timer, baseline=None):
    """
    Compare timer marks against baseline thresholds.
    Returns: list of (phase, actual_ms, threshold_ms, passed) tuples
    """
    baseline = baseline or DEFAULT_BASELINE
    results  = []
    for phase, threshold_ms in baseline.items():
        if phase in timer._marks:
            actual_ms = timer.elapsed_ms("start", phase)
            passed    = actual_ms <= threshold_ms if actual_ms else False
            results.append((phase, actual_ms, threshold_ms, passed))
    return results

if __name__ == "__main__":
    timer, result = timed_connect("CorpWifi-5G", "password123")
    if result["success"]:
        timer.report()
        checks = check_against_baseline(timer)
        print("\n[Baseline Check]")
        for phase, actual, threshold, ok in checks:
            icon = "✓" if ok else "✗"
            print(f"  [{icon}] {phase:25s}  {actual:7.1f}ms (max {threshold}ms)")
    else:
        print(f"[FAIL] {result['error']}")
