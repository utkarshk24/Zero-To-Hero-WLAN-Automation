The 'Why' — What This Capstone Builds

Five Stages, One Command

A professional WLAN connection acceptance test covers five distinct stages: (1) Scan — confirm the AP is visible and meets minimum RSSI. (2) Connect — establish association and 4-way handshake. (3) DHCP — verify a valid IP is assigned in the correct subnet. (4) DNS — confirm canary domains resolve correctly. (5) Connectivity — verify end-to-end internet reachability.

Exit code 0 = all 5 stages passed. Exit code 1 = any stage failed.
This is the standard CI convention. Your GitHub Actions workflow can run this tool and automatically fail the build if any stage fails — no human interpretation required. Starting Day 37, you'll wire this directly into a CI pipeline.

# Usage: python Day26_capstone.py --ssid CorpWifi-5G --password secret

import sys, json, subprocess, re, time, datetime, netifaces
import argparse
from pathlib import Path

def run_stage(name, func, *args, **kwargs):
    """
    Execute a named validation stage.
    Returns (stage_name, passed, details_dict, error_str).
    Catches all exceptions so one bad stage doesn't crash the tool.
    """
    print(f"\n[Stage] {name}...", end=" ", flush=True)
    try:
        result = func(*args, **kwargs)
        passed = result.get("passed", False)
        print("PASS ✓" if passed else "FAIL ✗")
        return name, passed, result, None
    except Exception as e:
        print(f"ERROR: {e}")
        return name, False, {}, str(e)
Block 2 — Stage Functions (Scan, Connect, DHCP, DNS, Connectivity)
Python 
def stage_scan(ssid, interface, min_rssi=-75):
    """Stage 1: Verify target SSID is visible with adequate signal."""
    result = subprocess.run(
        ["nmcli", "--terse", "--fields", "SSID,SIGNAL",
         "dev", "wifi", "list", "--rescan", "yes"],
        capture_output=True, text=True, timeout=30
    )
    for line in result.stdout.splitlines():
        parts = line.split(":")
        if parts[0] == ssid and len(parts) >= 2:
            signal = int(parts[1]) if parts[1].isdigit() else 0
            rssi   = (signal / 2) - 100   # Convert nmcli % to approx dBm
            passed = rssi >= min_rssi
            return {"passed": passed, "ssid": ssid,
                    "rssi_approx": rssi, "min_rssi": min_rssi}
    return {"passed": False, "error": f"'{ssid}' not found in scan"}

def stage_connect(ssid, password, interface):
    """Stage 2: Connect to the AP using nmcli."""
    subprocess.run(["nmcli", "dev", "disconnect", interface], capture_output=True)
    result = subprocess.run(
        ["nmcli", "--wait", "30", "dev", "wifi", "connect",
         ssid, "password", password, "ifname", interface],
        capture_output=True, text=True, timeout=35
    )
    return {"passed": result.returncode == 0,
            "detail": result.stdout.strip() or result.stderr.strip()}

def stage_dhcp(interface):
    """Stage 3: Validate DHCP assignment."""
    deadline = time.time() + 20
    while time.time() < deadline:
        try:
            addrs = netifaces.ifaddresses(interface)
            ip = addrs.get(netifaces.AF_INET, [{}])[0].get("addr", "")
            if ip and not ip.startswith("169.254."):
                gws = netifaces.gateways().get("default", {}).get(netifaces.AF_INET)
                gw  = gws[0] if gws and gws[1] == interface else None
                return {"passed": True, "ip": ip, "gateway": gw}
        except:
            pass
        time.sleep(0.5)
    return {"passed": False, "error": "DHCP timeout — no valid IP"}

def stage_dns():
    """Stage 4: Test DNS resolution."""
    import dns.resolver
    canaries = ["google.com", "cloudflare.com"]
    ok_count = 0
    for domain in canaries:
        try:
            dns.resolver.resolve(domain, "A", lifetime=5)
            ok_count += 1
        except:
            pass
    passed = ok_count == len(canaries)
    return {"passed": passed, "resolved": ok_count, "total": len(canaries)}

def stage_connectivity():
    """Stage 5: Verify end-to-end connectivity via ping."""
    result = subprocess.run(
        ["ping", "-c", "3", "-W", "3", "8.8.8.8"],
        capture_output=True
    )
    return {"passed": result.returncode == 0}
Block 3 — main() and JSON Report
Python 
def main():
    parser = argparse.ArgumentParser(description="WLAN Connection Validator")
    parser.add_argument("--ssid",      required=True)
    parser.add_argument("--password",  required=True)
    parser.add_argument("--interface", default="wlan0")
    parser.add_argument("--output",    default="reports")
    args = parser.parse_args()

    print(f"[WLAN Validator] Target: {args.ssid} on {args.interface}")
    print("=" * 50)

    stages = []
    all_passed = True

    # Stage 1: Scan
    name, ok, details, err = run_stage(
        "1. Scan", stage_scan, args.ssid, args.interface)
    stages.append({"stage": name, "passed": ok, "details": details})
    if not ok:
        all_passed = False
        print(f"  → Scan failed. Cannot proceed.")
        # Continue anyway for partial reporting

    # Stage 2: Connect (only if scan passed)
    if ok:
        name, ok, details, err = run_stage(
            "2. Connect", stage_connect, args.ssid, args.password, args.interface)
        stages.append({"stage": name, "passed": ok, "details": details})
        if not ok:
            all_passed = False

    # Stage 3: DHCP
    name, ok, details, err = run_stage(
        "3. DHCP", stage_dhcp, args.interface)
    stages.append({"stage": name, "passed": ok, "details": details})
    if not ok:
        all_passed = False

    # Stage 4: DNS
    name, ok, details, err = run_stage("4. DNS", stage_dns)
    stages.append({"stage": name, "passed": ok, "details": details})
    if not ok:
        all_passed = False

    # Stage 5: Connectivity
    name, ok, details, err = run_stage("5. Connectivity", stage_connectivity)
    stages.append({"stage": name, "passed": ok, "details": details})
    if not ok:
        all_passed = False

    # Write JSON report
    report = {
        "timestamp": datetime.datetime.now().isoformat(),
        "ssid": args.ssid, "interface": args.interface,
        "overall": "PASS" if all_passed else "FAIL",
        "stages": stages
    }
    out_dir = Path(args.output)
    out_dir.mkdir(exist_ok=True)
    ts      = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    report_path = out_dir / f"connection_report_{ts}.json"
    with open(report_path, "w") as f:
        json.dump(report, f, indent=2)

    print(f"\n{'='*50}")
    print(f"  OVERALL: {'PASS ✓' if all_passed else 'FAIL ✗'}")
    print(f"  Report : {report_path}")
    print(f"{'='*50}")

    sys.exit(0 if all_passed else 1)

if __name__ == "__main__":
    main()
