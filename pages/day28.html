The 'Why' — Conceptual Breakdown

From Scripts to Test Cases

Your Phase 3 scripts already validate RSSI and security. The difference with pytest is structure: each validation criterion becomes a named test with a clear pass/fail result, a docstring that explains what it tests and why, and an assertion message that tells you exactly what failed when something goes wrong.

Good test names are documentation.
test_all_aps_above_minimum_rssi tells a new engineer exactly what the test verifies without reading the code. When it fails in a CI report, the failure message immediately communicates the problem: "3 APs below -75 dBm threshold." This is the difference between a test that passes CI and one that actually helps engineers diagnose problems.

Block 1 — test_all_aps_above_minimum_rssi()

The assertion message in pytest is written after a comma in the assert statement: assert condition, "message". When the assertion fails, pytest prints this message alongside the variable values — essential for debugging in CI.

Python test_wlan_compliance.py 
# =============================================================
# Day 28: WLAN Test Cases with PyTest
# Run with: pytest test_wlan_compliance.py -v
# =============================================================

import pytest

# Simulated AP scan results (replace with real scan in Day 30)
SIMULATED_SCAN = [
    {"ssid": "CorpWifi-5G",   "bssid": "AA:BB:CC:01:02:03",
     "rssi": -52, "channel": 36, "security": "WPA2"},
    {"ssid": "CorpWifi-2G",   "bssid": "AA:BB:CC:01:02:04",
     "rssi": -68, "channel": 6,  "security": "WPA2"},
    {"ssid": "GuestWifi",     "bssid": "AA:BB:CC:01:02:05",
     "rssi": -71, "channel": 11, "security": "WPA2"},
]

# Expected APs from inventory
EXPECTED_BSSIDS = {"AA:BB:CC:01:02:03", "AA:BB:CC:01:02:04", "AA:BB:CC:01:02:05"}
MIN_RSSI        = -75

def test_all_aps_above_minimum_rssi():
    """
    All discovered APs must have RSSI above the site minimum threshold.
    Failing APs are listed in the assertion message for easy debugging.
    """
    failing_aps = [
        f"{ap['ssid']} ({ap['bssid']}): {ap['rssi']} dBm"
        for ap in SIMULATED_SCAN
        if ap["rssi"] < MIN_RSSI
    ]
    assert len(failing_aps) == 0, (
        f"APs below {MIN_RSSI} dBm threshold:\n  " + "\n  ".join(failing_aps)
    )

def test_expected_ap_count():
    """The number of discovered APs must match the inventory."""
    discovered_bssids = {ap["bssid"] for ap in SIMULATED_SCAN}
    assert len(discovered_bssids) == len(EXPECTED_BSSIDS), (
        f"Expected {len(EXPECTED_BSSIDS)} APs, found {len(discovered_bssids)}"
    )
Block 2 — TestRogueAPDetection Class

pytest supports both function-based tests and class-based tests. Classes are useful for grouping related tests that share setup context. No subclassing or inheritance needed — just a class named Test*.

Python 
class TestRogueAPDetection:
    """Tests for detecting unauthorised APs in the RF environment."""

    def test_no_rogue_bssids(self):
        """
        All discovered APs must have BSSIDs in the approved inventory.
        Any unknown BSSID is a potential rogue AP.
        """
        discovered_bssids = {ap["bssid"] for ap in SIMULATED_SCAN}
        rogue_bssids = discovered_bssids - EXPECTED_BSSIDS
        assert len(rogue_bssids) == 0, (
            f"Rogue APs detected with unknown BSSIDs: {rogue_bssids}"
        )

    def test_no_evil_twin_ssids(self):
        """
        Check for APs with our corporate SSID names but unknown BSSIDs
        (evil twin attack pattern).
        """
        corp_ssids     = {"CorpWifi-5G", "CorpWifi-2G", "GuestWifi"}
        evil_twins     = []
        for ap in SIMULATED_SCAN:
            if ap["ssid"] in corp_ssids and ap["bssid"] not in EXPECTED_BSSIDS:
                evil_twins.append(f"{ap['ssid']} from {ap['bssid']}")
        assert len(evil_twins) == 0, (
            f"Potential evil twin APs:\n  " + "\n  ".join(evil_twins)
        )

    def test_no_open_ssids_matching_corporate_names(self):
        """Open security on a corporate SSID name is an attack vector."""
        violations = [
            f"{ap['ssid']} ({ap['bssid']})"
            for ap in SIMULATED_SCAN
            if ap["ssid"].startswith("Corp") and ap["security"] == "Open"
        ]
        assert len(violations) == 0, f"Open security on Corp SSID: {violations}"
Block 3 — Channel Compliance with @pytest.mark.parametrize

@pytest.mark.parametrize generates multiple test cases from a single function. Instead of writing test_channel_1(), test_channel_6(), etc., you write one function and parametrize it. pytest names each case automatically.

Python 
NON_OVERLAPPING_2GHZ = {1, 6, 11}

def test_non_overlapping_channels():
    """
    All 2.4 GHz APs must use non-overlapping channels (1, 6, or 11).
    Adjacent channel interference causes retransmissions and poor performance.
    """
    violations = []
    for ap in SIMULATED_SCAN:
        ch = ap["channel"]
        if ch <= 14 and ch not in NON_OVERLAPPING_2GHZ:
            violations.append(f"{ap['ssid']} on channel {ch}")
    assert len(violations) == 0, (
        f"APs on overlapping 2.4 GHz channels:\n  " + "\n  ".join(violations)
    )

@pytest.mark.parametrize("ssid,expected_security", [
    ("CorpWifi-5G",  "WPA2"),
    ("CorpWifi-2G",  "WPA2"),
    ("GuestWifi",    "WPA2"),
])
def test_ssid_security(ssid, expected_security):
    """Each SSID must use the expected security type."""
    for ap in SIMULATED_SCAN:
        if ap["ssid"] == ssid:
            assert ap["security"] == expected_security, (
                f"'{ssid}' uses '{ap['security']}' but expected '{expected_security}'"
            )
            return
    pytest.skip(f"SSID '{ssid}' not found in scan — skipping security check")
