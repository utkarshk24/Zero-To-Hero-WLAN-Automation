The 'Why' — What This Capstone Builds

Professional Test Suite Structure

A flat folder of test files doesn't scale. When you have 50+ tests, you need structure: tests grouped by feature, shared fixtures in conftest.py, a central configuration in pytest.ini, and mark-based test selection for different contexts (smoke tests in 30 seconds, full regression in 10 minutes).

The directory structure is the architecture.
tests/ ── test_rssi.py, test_security.py, test_channels.py, test_connectivity.py, conftest.py, pytest.ini. This structure tells every engineer where to find what, and tells pytest exactly how to discover, configure, and report tests.

# tests/conftest.py — Session fixtures and hooks for the full suite

import pytest
import json
import yaml
import datetime
from pathlib import Path

def pytest_configure(config):
    """Add custom metadata to HTML report header."""
    config._metadata = getattr(config, "_metadata", {})
    config._metadata["Test Suite"] = "WLAN Automation Capstone"
    config._metadata["Timestamp"]  = datetime.datetime.now().isoformat()

@pytest.fixture(scope="session")
def ap_inventory():
    """Load AP inventory. Skip session if file is missing."""
    path = Path("../ap_inventory.json")
    if not path.exists():
        pytest.skip("ap_inventory.json not found")
    return json.loads(path.read_text())

@pytest.fixture(scope="session")
def wlan_config():
    """Load YAML test thresholds."""
    path = Path("../test_config.yaml")
    if not path.exists():
        return {"thresholds": {"min_rssi_dbm": -75}}
    return yaml.safe_load(path.read_text())

@pytest.fixture(scope="session")
def scan_data():
    """
    Return scan data. In CI/hardware tests, replace with real scan.
    Marked as session scope so scan runs once per pytest invocation.
    """
    return [
        {"ssid": "CorpWifi-5G", "bssid": "AA:BB:CC:01:02:03",
         "rssi": -52, "channel": 36, "security": "WPA2"},
        {"ssid": "CorpWifi-2G", "bssid": "AA:BB:CC:01:02:04",

Block 4 — Sample Test Files with Marks
Python tests/test_rssi.py 
# tests/test_rssi.py
import pytest
import allure

@allure.feature("RSSI Compliance")
class TestRSSICompliance:

    @pytest.mark.smoke
    @pytest.mark.rssi
    @allure.severity(allure.severity_level.CRITICAL)
    def test_all_aps_above_threshold(self, scan_data, wlan_config):
        """SMOKE: All APs must meet minimum RSSI threshold."""
        threshold = wlan_config["thresholds"]["min_rssi_dbm"]
        failing = [ap for ap in scan_data if ap["rssi"] < threshold]
        assert len(failing) == 0, (
            f"APs below {threshold} dBm: "
            + ", ".join(f"{a['ssid']}={a['rssi']}" for a in failing)
        )

    @pytest.mark.smoke
    @pytest.mark.rssi
    def test_strongest_ap_above_minus60(self, scan_data):
        """SMOKE: At least one AP should be stronger than -60 dBm."""
        strongest = max(scan_data, key=lambda a: a["rssi"])
        assert strongest["rssi"] >= -60, (
            f"Strongest AP '{strongest['ssid']}' is only {strongest['rssi']} dBm"
        )

# Run only smoke tests:  pytest tests/ -m smoke
# Run only RSSI tests:   pytest tests/ -m rssi
# Run all:               pytest tests/
