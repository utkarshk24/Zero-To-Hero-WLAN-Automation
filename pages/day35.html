The 'Why' — Conceptual Breakdown

Why RSSI Isn't Enough

An RSSI of -60 dBm suggests good signal quality. But that same -60 dBm AP might only deliver 5 Mbps of actual throughput if the channel is congested with interference from neighbouring APs, or if the client's rate selection algorithm is choosing a conservative MCS rate. Throughput testing cuts through all these variables and measures what users actually experience.

iPerf3 --json output makes parsing trivial.
iperf3 -c server -J outputs a single JSON object containing the complete test result: start time, intervals, summary stats, and error messages. json.loads(result.stdout) gives you a Python dict with everything you need — no regex, no string parsing.

TCP vs. UDP Testing

TCP throughput measures how fast a real file transfer would go — TCP's congestion control and retransmission mechanisms mean the measured Mbps directly reflects the usable bandwidth for applications. UDP testing (with -u -b 100M) measures jitter and packet loss — the metrics that matter for VoIP, video conferencing, and real-time applications that can't retransmit lost packets.

Block 1 — run_iperf3() with JSON Output Parsing
Python Day35_iperf3.py 
# =============================================================
# Day 35: Throughput Testing with iPerf3
# Requires: iperf3 installed + iperf3 server running on network
# =============================================================

import subprocess
import json

def run_iperf3(server, duration=10, protocol="tcp", bitrate=None,
               reverse=False, port=5201):
    """
    Run an iPerf3 test and return parsed results.

    Args:
        server:   iPerf3 server IP or hostname
        duration: Test duration in seconds (default 10)
        protocol: 'tcp' or 'udp'
        bitrate:  Target bitrate for UDP (e.g. '100M' = 100 Mbps)
        reverse:  If True, measure download (server→client)
        port:     iPerf3 server port (default 5201)

    Returns:
        dict with: mbps, jitter_ms, lost_pct, retransmits, error
    """
    cmd = ["iperf3", "-c", server, "-p", str(port),
           "-t", str(duration), "-J"]   # -J = JSON output

    if protocol == "udp":
        cmd.extend(["-u"])
        if bitrate:
            cmd.extend(["-b", bitrate])   # UDP must specify target bitrate

    if reverse:
        cmd.append("-R")   # Reverse = download test (server sends to client)

    result = subprocess.run(cmd, capture_output=True, text=True, timeout=duration + 30)

    if result.returncode != 0:
        return {"error": result.stderr.strip() or "iperf3 failed", "mbps": 0}

    try:
        data = json.loads(result.stdout)
    except json.JSONDecodeError:
        return {"error": "Failed to parse iperf3 JSON output", "mbps": 0}

    # Navigate the iPerf3 JSON structure
    end = data.get("end", {})
    streams = end.get("streams", [{}])
    summary = end.get("sum_sent", end.get("sum", {}))

    # TCP: bits_per_second in sum_sent
    # UDP: bits_per_second + jitter + lost_percent in sum
    mbps = summary.get("bits_per_second", 0) / 1e6
    jitter    = summary.get("jitter_ms", 0)
    lost_pct  = summary.get("lost_percent", 0)
    retrans   = summary.get("retransmits", 0)

    return {
        "mbps":       round(mbps, 2),
        "jitter_ms":  round(jitter, 3),
        "lost_pct":   round(lost_pct, 2),
        "retransmits": retrans,
        "duration_s": duration,
        "protocol":   protocol,
        "reverse":    reverse,
        "error":      None,
    }

Block 2 — bidirectional_test()

Bidirectional testing (upload + download) is more comprehensive than single direction. Most WLAN issues affect asymmetric traffic differently — a congested downlink might still have a clean uplink.

Python 
def bidirectional_test(server, duration=10):
    """
    Run both upload and download TCP iPerf3 tests.
    Returns a summary dict with both directions.
    """
    print(f"[iPerf3] Upload test ({duration}s)...")
    upload = run_iperf3(server, duration=duration, reverse=False)

    print(f"[iPerf3] Download test ({duration}s)...")
    download = run_iperf3(server, duration=duration, reverse=True)

    summary = {
        "upload_mbps":   upload.get("mbps", 0),
        "download_mbps": download.get("mbps", 0),
        "upload_retrans": upload.get("retransmits", 0),
        "download_retrans": download.get("retransmits", 0),
    }

    print(f"\n[Throughput Summary]")
    print(f"  Upload   : {summary['upload_mbps']:6.1f} Mbps  "
          f"(retransmits: {summary['upload_retrans']})")
    print(f"  Download : {summary['download_mbps']:6.1f} Mbps  "
          f"(retransmits: {summary['download_retrans']})")

    return summary
Block 3 — udp_quality_test() for Jitter and Loss

UDP testing at a fixed bitrate reveals jitter and packet loss — metrics that matter for real-time applications like video conferencing and VoIP.

Python 
def udp_quality_test(server, bitrate="50M", duration=10):
    """
    Run a UDP iPerf3 test to measure jitter and packet loss.
    These metrics directly predict VoIP/video conferencing quality.
    """
    print(f"[iPerf3] UDP quality test at {bitrate} ({duration}s)...")
    result = run_iperf3(server, duration=duration, protocol="udp",
                         bitrate=bitrate)

    print(f"\n[UDP Quality]")
    print(f"  Jitter       : {result.get('jitter_ms', '?'):.3f} ms")
    print(f"  Packet loss  : {result.get('lost_pct', '?'):.2f}%")
    print(f"  Actual Mbps  : {result.get('mbps', '?'):.1f}")

    return result

import pytest

@pytest.mark.integration
def test_throughput_meets_sla():
    """Upload and download must both exceed 50 Mbps for 802.11ac deployment."""
    SERVER = "192.168.1.1"   # Replace with your iPerf3 server IP
    result = bidirectional_test(SERVER, duration=10)
    assert result["upload_mbps"]   >= 50, f"Upload {result['upload_mbps']:.1f} Mbps below 50 Mbps SLA"
    assert result["download_mbps"] >= 50, f"Download {result['download_mbps']:.1f} Mbps below 50 Mbps SLA"

@pytest.mark.integration
def test_udp_jitter_acceptable():
    """UDP jitter must be below 10ms for VoIP quality."""
    SERVER = "192.168.1.1"
    result = udp_quality_test(SERVER, bitrate="10M", duration=10)
    assert result.get("jitter_ms", 999) < 10, f"Jitter {result.get('jitter_ms')} ms exceeds 10ms"
    assert result.get("lost_pct", 100) < 1.0, f"Packet loss {result.get('lost_pct')}% exceeds 1%"
