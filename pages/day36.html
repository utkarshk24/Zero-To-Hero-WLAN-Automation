1 — Detecting Roaming Capability

python 

import subprocess, re

def check_roaming_capabilities(interface="wlan0"):
    """Check which 802.11r/k/v capabilities are active on the current connection."""
    result = subprocess.run(
        ["iw", "dev", interface, "link"],
        capture_output=True, text=True
    )
    output = result.stdout.lower()
    caps = {
        "802.11r_ft":  "ft" in output or "fast-transition" in output,
        "802.11k_rrm": "rrm" in output or "radio-resource" in output,
        "802.11v_btm": "bss-trans" in output or "btm" in output,
    }
    for cap, supported in caps.items():
        print(f"  {cap}: {'Supported' if supported else 'Not detected'}")
    return caps


def get_current_bssid(interface="wlan0"):
    """Return the BSSID of the currently associated AP."""
    r = subprocess.run(["iw", "dev", interface, "link"], capture_output=True, text=True)
    m = re.search(r"Connected to ([0-9a-f:]{17})", r.stdout, re.IGNORECASE)
    return m.group(1).upper() if m else None

2 — Measuring Roaming Latency

python 

import time, threading

def measure_roaming_latency(interface="wlan0", target_bssid=None, timeout=10):
    """
    Measure time from current BSSID → new BSSID.
    Returns latency_ms or None on timeout.
    """
    original_bssid = get_current_bssid(interface)
    if not original_bssid:
        return None, "Not connected"

    print(f"Monitoring roam from {original_bssid}...")
    start = time.perf_counter()
    deadline = start + timeout

    while time.perf_counter() < deadline:
        current = get_current_bssid(interface)
        if current and current != original_bssid:
            if target_bssid is None or current == target_bssid.upper():
                latency = (time.perf_counter() - start) * 1000
                return round(latency, 1), None
        time.sleep(0.01)   # 10 ms poll

    return None, f"No roam detected within {timeout}s"


def trigger_roam_via_btm(interface="wlan0", target_bssid=None):
    """Send BSS Transition Management request to trigger a roam (wpa_cli)."""
    cmd = ["wpa_cli", "-i", interface, "roam"]
    if target_bssid:
        cmd.append(target_bssid)
    subprocess.run(cmd, capture_output=True)

3 — Roaming Test with PyTest

python 

import pytest, allure

ROAMING_LATENCY_LIMIT_MS = 150   # Wi-Fi 6 typical target: <50 ms, conservative: <150 ms

@allure.feature("Roaming")
@allure.severity(allure.severity_level.CRITICAL)
@pytest.mark.live
def test_roaming_latency():
    """Roaming must complete within the latency limit."""
    before = get_current_bssid("wlan0")
    allure.attach(f"Source BSSID: {before}", name="Pre-roam state")

    trigger_roam_via_btm("wlan0")
    latency, err = measure_roaming_latency("wlan0", timeout=5)

    after = get_current_bssid("wlan0")
    allure.attach(f"Target BSSID: {after}\nLatency: {latency} ms", name="Post-roam state")

    assert err is None, f"Roaming failed: {err}"
    assert latency <= ROAMING_LATENCY_LIMIT_MS, \
        f"Roaming too slow: {latency} ms > {ROAMING_LATENCY_LIMIT_MS} ms"
    assert before != after, "BSSID did not change — no roaming occurred"
