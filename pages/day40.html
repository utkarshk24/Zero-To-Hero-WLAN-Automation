The Core Framework Module (src/validator.py)

python 

"""validator.py — The heart of the framework."""

import json, datetime, sys
from config import load_config
from scanner import scan_networks, find_ssid
from connector import connect_wpa2, disconnect
from wlan_utils import validate_dhcp, dns_health_check

class WLANValidator:
    def __init__(self):
        self.config  = load_config()
        self.results = []
        self.start   = datetime.datetime.now()

    def run(self, ssid=None, password=None):
        ssid     = ssid     or self.config["WIFI_SSID"]
        password = password or self.config["WIFI_PASSWORD"]

        print(f"\n{'='*60}")
        print(f" WLAN Validation Framework")
        print(f" Target: {ssid}  |  {self.start.strftime('%Y-%m-%d %H:%M')}")
        print(f"{'='*60}\n")

        stages = [
            ("Scan",          self._stage_scan,    ssid),
            ("Connect",       self._stage_connect, ssid, password),
            ("DHCP",          self._stage_dhcp),
            ("DNS",           self._stage_dns),
            ("Connectivity",  self._stage_ping),
        ]
        for name, fn, *args in stages:
            result = fn(*args)
            result["stage"] = name
            self.results.append(result)
            icon = "✓" if result["passed"] else "✗"
            detail = result.get("detail", "")
            print(f"  [{icon}] {name:15s}  {detail}")

        self._save_report()
        passed = all(r["passed"] for r in self.results)
        print(f"\n{'PASS' if passed else 'FAIL'}: {sum(r['passed'] for r in self.results)}/{len(self.results)} stages")
        return 0 if passed else 1

    def _stage_scan(self, ssid):
        nets = scan_networks()
        match = find_ssid(ssid, nets)
        rssi  = match[0]["rssi"] if match else None
        return {"passed": bool(match) and (rssi or 0) >= self.config["MIN_RSSI"],
                "detail": f"RSSI={rssi} dBm, found={len(match)} instance(s)"}

    def _stage_connect(self, ssid, password):
        ok, elapsed, err = connect_wpa2(ssid, password)
        return {"passed": ok, "detail": f"{elapsed:.2f}s  {err or ''}"}

    def _stage_dhcp(self):
        ok = validate_dhcp(self.config["AP_INTERFACE"])
        return {"passed": ok, "detail": "IP assigned" if ok else "No DHCP IP"}

    def _stage_dns(self):
        ok = dns_health_check()
        return {"passed": ok, "detail": "DNS resolves" if ok else "DNS failed"}

    def _stage_ping(self):
        import subprocess, netifaces
        gw = netifaces.gateways().get("default", {}).get(2, [None])[0]
        ok = subprocess.run(["ping", "-c", "3", gw or "8.8.8.8"],
                            capture_output=True).returncode == 0
        return {"passed": ok, "detail": f"Gateway {gw}"}

    def _save_report(self):
        report = {
            "timestamp": self.start.isoformat(),
            "stages":    self.results,
            "passed":    all(r["passed"] for r in self.results),
        }
        path = f"reports/validation_{self.start.strftime('%Y%m%d_%H%M%S')}.json"
        with open(path, "w") as f:
            json.dump(report, f, indent=2)
        print(f"\n  Report: {path}")


if __name__ == "__main__":
    sys.exit(WLANValidator().run())
